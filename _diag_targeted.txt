function safeResourceUrl(value, kind='image'){
  const raw = String(value == null ? '' : value).trim();
  if(!raw || /[\u0000-\u001f\u007f]/.test(raw)) return '';
  if(/^blob:/i.test(raw) || /^https?:\/\//i.test(raw) || /^(file|content):/i.test(raw)) return raw;
  if(kind === 'video' && /^data:video\/[a-z0-9.+-]+(?:;[^,]*)?,/i.test(raw)) return raw;
  if(kind === 'audio' && /^data:audio\/[a-z0-9.+-]+(?:;[^,]*)?,/i.test(raw)) return raw;
  if(/^data:image\/[a-z0-9.+-]+(?:;[^,]*)?,/i.test(raw)) return raw;
  return '';
}

      const img = new Image();
      img.onload = () => {
        let w=img.width,h=img.height;

    };

  });
}
